# ============================================================================
#  WordPress .htaccess - baseline with hardening
#
#  Place at:  the webroot of the installation
#             - standard install:     public_html/.htaccess
#             - subdirectory layout:  public_html/.htaccess  (this file)
#               plus public_html/wordpress/.htaccess (see below)
#
#  IMPORTANT - ordering:
#  WordPress manages the block between "# BEGIN WordPress" and
#  "# END WordPress" itself and overwrites it whenever permalink settings are
#  saved. Everything of your own belongs BEFORE or AFTER it, never inside.
#
#  Requirement: AllowOverride All for this directory. Without it the whole
#  file is silently ignored:
#      grep -rh AllowOverride /etc/apache2/sites-enabled/
#
#  Check after every change:
#      apachectl configtest && systemctl reload apache2
#      curl -sI https://YOUR-DOMAIN/ | head -20
# ============================================================================


# ----------------------------------------------------------------------------
# 1. Disable directory listings
#    Without an index file Apache would otherwise list the directory contents.
# ----------------------------------------------------------------------------
Options -Indexes

# NOTE: every Options directive requires "AllowOverride Options" or "All".
# Panels (Virtualmin, Plesk, cPanel) usually set a whitelist that does not
# include FollowSymLinks. Apache then answers with
#     Option FollowSymLinks not allowed here
# and returns 500 for everything below - including CSS and JS.
#
# That is why only -Indexes is used here. If your vhost allows more, you can
# add (test first!):
#     Options +SymLinksIfOwnerMatch -Includes -ExecCGI
# Check what is permitted:
#     grep -rh AllowOverride /etc/apache2/sites-enabled/


# ----------------------------------------------------------------------------
# 2. Block sensitive files
#    These are either configuration or reveal version information.
# ----------------------------------------------------------------------------
<FilesMatch "^(wp-config\.php|wp-config-sample\.php|\.htaccess|\.htpasswd|\.user\.ini|php\.ini|readme\.html|license\.txt|liesmich\.html|wp-config\.php\.bak.*)$">
    Require all denied
</FilesMatch>

# Debug logs, database dumps, archives, backup copies, version control.
# Such files do not belong in the webroot - but if one is left behind, at
# least it is not retrievable.
<FilesMatch "\.(sql|sql\.gz|tar|tar\.gz|tgz|zip|bak|bakup|backup|old|orig|save|swp|swo|log|phar|env|ini|conf|sh|dist|inc)$">
    Require all denied
</FilesMatch>

# Dot files (.git, .env, .DS_Store ...) - except the ACME directory that
# Let's Encrypt needs for domain validation.
RedirectMatch 404 /\.(?!well-known/)

# Version control and package manager directories
RedirectMatch 404 /(\.git|\.svn|\.hg|node_modules|vendor/bin)/


# ----------------------------------------------------------------------------
# 3. Prevent PHP execution where it has no business
#
#    The single most important point in this file. An uploaded PHP file in
#    uploads/ is the classic route to a backdoor. In addition, a dedicated
#    .htaccess belongs directly in wp-content/uploads/ - see the end of this
#    file.
# ----------------------------------------------------------------------------
<FilesMatch "\.(php|php[0-9]|phtml|phps|pht|phar|shtml|cgi|pl|py|asp|aspx|exe)$">
    <If "%{REQUEST_URI} =~ m#/wp-content/(uploads|cache|languages)/#">
        Require all denied
    </If>
</FilesMatch>


# ----------------------------------------------------------------------------
# 4. XML-RPC
#
#    Needed for pingbacks, the classic WordPress app and Jetpack. If you use
#    none of those, the file is a pure attack surface: system.multicall lets
#    hundreds of login attempts be bundled into a single request.
#
#    Enabled by default. Comment the block out if you use the WordPress app
#    or Jetpack.
# ----------------------------------------------------------------------------
<Files "xmlrpc.php">
    Require all denied
</Files>


# ----------------------------------------------------------------------------
# 5. Security headers
#    These complement the browser's own protections. Content-Security-Policy
#    is deliberately absent - a useful CSP depends on the theme and plugins
#    and has to be worked out individually, otherwise it breaks the site.
# ----------------------------------------------------------------------------
<IfModule mod_headers.c>
    # Prevent MIME type sniffing (e.g. an HTML file disguised as an image)
    Header always set X-Content-Type-Options "nosniff"

    # The site must not be embedded in foreign frames (clickjacking)
    Header always set X-Frame-Options "SAMEORIGIN"

    # Send the full referrer only to the same origin
    Header always set Referrer-Policy "strict-origin-when-cross-origin"

    # Turn off device access a website normally does not need
    Header always set Permissions-Policy "geolocation=(), microphone=(), camera=(), payment=(), usb=()"

    # HSTS - only enable once HTTPS works reliably on ALL subdomains.
    # There is no way back during max-age.
    # Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains"

    # Do not advertise the PHP version
    Header always unset X-Powered-By
    Header always unset X-Pingback
</IfModule>

# Reduce the server signature (only effective if not overridden globally)
ServerSignature Off


# ----------------------------------------------------------------------------
# 6. Make user enumeration harder
#    ?author=1 normally redirects to /author/username/ and thereby reveals the
#    login name. The REST endpoint /wp-json/wp/v2/users does the same.
#
#    Note: blocking the REST endpoint is safe only as long as no plugin needs
#    it publicly. The block editor keeps using it while logged in, and the
#    rule cannot tell the two apart - so only the author query is handled here.
# ----------------------------------------------------------------------------
<IfModule mod_rewrite.c>
    RewriteEngine On

    # /?author=123
    RewriteCond %{QUERY_STRING} (^|&)author=\d+ [NC]
    RewriteRule ^$ - [F,L]
</IfModule>


# ----------------------------------------------------------------------------
# 7. Compression and browser caching
#    Not a security topic, but the biggest lever for load time.
# ----------------------------------------------------------------------------
<IfModule mod_deflate.c>
    AddOutputFilterByType DEFLATE text/plain text/html text/xml text/css
    AddOutputFilterByType DEFLATE application/xml application/xhtml+xml
    AddOutputFilterByType DEFLATE application/javascript application/x-javascript
    AddOutputFilterByType DEFLATE application/json application/rss+xml
    AddOutputFilterByType DEFLATE image/svg+xml font/woff2 font/woff
</IfModule>

<IfModule mod_expires.c>
    ExpiresActive On
    ExpiresDefault                          "access plus 1 month"
    ExpiresByType text/html                 "access plus 0 seconds"
    ExpiresByType application/json          "access plus 0 seconds"
    ExpiresByType text/xml                  "access plus 0 seconds"
    ExpiresByType image/jpeg                "access plus 1 year"
    ExpiresByType image/png                 "access plus 1 year"
    ExpiresByType image/webp                "access plus 1 year"
    ExpiresByType image/avif                "access plus 1 year"
    ExpiresByType image/svg+xml             "access plus 1 year"
    ExpiresByType font/woff2                "access plus 1 year"
    ExpiresByType text/css                  "access plus 1 year"
    ExpiresByType application/javascript    "access plus 1 year"
</IfModule>

# Character set, so that special characters arrive correctly without a meta tag
AddDefaultCharset UTF-8


# ============================================================================
# BEGIN WordPress
# The directives between "BEGIN WordPress" and "END WordPress" are generated
# dynamically and should only be modified via WordPress filters. Changes here
# will be overwritten.
# ============================================================================
<IfModule mod_rewrite.c>
RewriteEngine On
RewriteRule .* - [E=HTTP_AUTHORIZATION:%{HTTP:Authorization}]
RewriteBase /
RewriteRule ^index\.php$ - [L]
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d
RewriteRule . /index.php [L]
</IfModule>
# END WordPress


# ============================================================================
#  ADDITIONAL FILES - create separately, do not paste here
# ============================================================================
#
#  wp-content/uploads/.htaccess
#  ----------------------------
#  Second line of defence for the most important point. Even though section 3
#  covers it: belt and braces, and this file keeps working if the main file is
#  ever overwritten.
#
#      <IfModule mod_mime.c>
#          RemoveHandler .php .php3 .php4 .php5 .php7 .php8 .php8.3 .phtml .phps .phar
#      </IfModule>
#      <FilesMatch "\.(php|php[0-9]|php[0-9]\.[0-9]|phtml|phps|pht|phar|shtml|cgi|pl|py)$">
#          Require all denied
#      </FilesMatch>
#      Options -Indexes
#
#  Do NOT use "AddType text/plain .php": with mod_fcgid that overrides the
#  handler mapping from the vhost and breaks PHP for the whole tree - uploads
#  then fail with "could not be moved".
#
#
#  wp-includes/.htaccess  (optional)
#  ---------------------------------
#  Prevent direct calls to core files. Careful: ms-files.php is needed for
#  multisite, wp-tinymce.php by the editor.
#
#      <FilesMatch "^(wp-tinymce\.php|ms-files\.php)$">
#          Require all granted
#      </FilesMatch>
#      <FilesMatch "\.(php|phtml)$">
#          Require all denied
#      </FilesMatch>
#
#
#  Subdirectory layout (core in public_html/wordpress/)
#  ----------------------------------------------------
#  This file stays in the webroot. The core directory gets its own .htaccess
#  with RewriteBase pointing at the subdirectory:
#
#      # BEGIN WordPress
#      <IfModule mod_rewrite.c>
#      RewriteEngine On
#      RewriteBase /wordpress/
#      RewriteRule ^index\.php$ - [L]
#      RewriteCond %{REQUEST_FILENAME} !-f
#      RewriteCond %{REQUEST_FILENAME} !-d
#      RewriteRule . /wordpress/index.php [L]
#      </IfModule>
#      # END WordPress
#
#
#  Protecting wp-admin (optional, more effective than anything else)
#  -----------------------------------------------------------------
#  With a static IP address:
#
#      <Files "wp-login.php">
#          Require ip 203.0.113.5
#      </Files>
#
#  With a changing IP, use basic auth in front of wp-login.php instead:
#
#      htpasswd -c /home/SITE/.htpasswd adminname
#
#      <Files "wp-login.php">
#          AuthType Basic
#          AuthName "Restricted"
#          AuthUserFile /home/SITE/.htpasswd
#          Require valid-user
#      </Files>
#
#  Careful: with basic auth active, application passwords and REST API
#  authentication stop working - WordPress reports this in Site Health.
#
#
#  What is deliberately NOT here
#  -----------------------------
#  - Blocking /wp-json/ entirely: breaks the block editor and many plugins
#  - "Content-Security-Policy": must be worked out per theme, otherwise a
#    blank page. Test in report-only mode first.
#  - Blocking user agents or countries: creates more false positives than
#    value and goes stale quickly
#  - php_value directives: these only work with mod_php, not with PHP-FPM.
#    There such settings belong in .user.ini
# ============================================================================
