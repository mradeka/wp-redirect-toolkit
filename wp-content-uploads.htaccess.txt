# ============================================================================
#  wp-content/uploads/.htaccess
#
#  Zweite Verteidigungslinie: verhindert die Ausfuehrung hochgeladener
#  Skripte, selbst wenn die Haupt-.htaccess ueberschrieben wird.
#
#  Ablage: /pfad/zur/installation/wp-content/uploads/.htaccess
#  Rechte: 644, Eigentuemer = Seitenbenutzer
# ============================================================================

# Keine Ausfuehrung von Skripten in diesem Verzeichnis und darunter
<FilesMatch "\.(php|php[0-9]|phtml|phps|pht|phar|shtml|cgi|pl|py|asp|aspx)$">
    Require all denied
</FilesMatch>

# Doppelte Endungen: "bild.php.jpg" endet zwar auf .jpg, kann aber auf
# Servern, die PHP ueber mod_mime anhand einer beliebigen Endung im Namen
# zuordnen, trotzdem ausgefuehrt werden. Deshalb jede Datei sperren, die
# eine Skript-Endung IRGENDWO im Namen traegt.
<FilesMatch "\.(php|php[0-9]|phtml|phps|pht|phar|shtml|cgi|pl|py)\.">
    Require all denied
</FilesMatch>

# Selbst wenn eine Datei durchrutscht: als Text ausliefern, nicht ausfuehren
AddType text/plain .php .phtml .phps .php3 .php4 .php5 .php7 .phar

# Handler-Zuordnung fuer PHP in diesem Verzeichnis vollstaendig aufheben
<IfModule mod_mime.c>
    RemoveHandler .php .phtml .phps .php3 .php4 .php5 .php7 .phar
    RemoveType    .php .phtml .phps .php3 .php4 .php5 .php7 .phar
</IfModule>

# Nur -Indexes: -ExecCGI und -Includes braeuchten "AllowOverride Options",
# das viele Panels nicht gewaehren - dann liefert Apache 500 fuer das ganze
# Verzeichnis. Die PHP-Ausfuehrung ist oben bereits ueber FilesMatch, AddType
# und RemoveHandler unterbunden; dafuer genuegt AllowOverride FileInfo.
Options -Indexes
