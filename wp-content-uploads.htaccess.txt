# ============================================================================
#  wp-content/uploads/.htaccess
#
#  Zweite Verteidigungslinie: verhindert die Ausfuehrung hochgeladener
#  Skripte, selbst wenn die Haupt-.htaccess ueberschrieben wird.
#
#  Ablage: /pfad/zur/installation/wp-content/uploads/.htaccess
#  Rechte: 644, Eigentuemer = Seitenbenutzer
# ============================================================================

# WICHTIG bei mod_fcgid / suexec (Virtualmin, Plesk, cPanel):
# Dort laeuft PHP ueber "AddHandler fcgid-script .php" aus dem vhost. Ein
# "AddType text/plain .php" hier ueberschreibt die Zuordnung und bricht die
# PHP-Verarbeitung im ganzen Verzeichnisbaum - Uploads im Backend scheitern
# dann mit "The uploaded file could not be moved to wp-content/uploads/...".
# Richtig ist RemoveHandler: es entfernt die Zuordnung nur fuer dieses
# Verzeichnis.
<IfModule mod_mime.c>
    RemoveHandler .php .php3 .php4 .php5 .php7 .php8 .php8.0 .php8.1 .php8.2 .php8.3 .php8.4 .phtml .phps .phar .pht
</IfModule>

# Zusaetzlich der direkte Zugriffsschutz. Nur Skript-Endungen - Bilder, PDFs
# und alles andere bleiben abrufbar.
<FilesMatch "\.(php|php[0-9]|php[0-9]\.[0-9]|phtml|phps|pht|phar|shtml|cgi|pl|py|asp|aspx)$">
    Require all denied
</FilesMatch>

# Doppelte Endungen ("bild.php.jpg") werden auf manchen Servern trotzdem als
# PHP ausgefuehrt - deshalb jede Skript-Endung IRGENDWO im Namen sperren.
<FilesMatch "\.(php|php[0-9]|phtml|phps|pht|phar|shtml|cgi|pl|py)\.">
    Require all denied
</FilesMatch>

# Nur -Indexes: -ExecCGI und -Includes braeuchten "AllowOverride Options",
# das viele Panels nur mit einer Whitelist gewaehren.
Options -Indexes
