# ============================================================================
#  wp-content/uploads/.htaccess
#
#  Second line of defence: prevents execution of uploaded scripts even if the
#  main .htaccess is overwritten.
#
#  Place at: /path/to/installation/wp-content/uploads/.htaccess
#  Mode:     644, owned by the site user
# ============================================================================

# IMPORTANT with mod_fcgid / suexec (Virtualmin, Plesk, cPanel):
# There PHP runs via "AddHandler fcgid-script .php" from the vhost. An
# "AddType text/plain .php" here overrides that mapping and breaks PHP
# processing for the whole directory tree - uploads in wp-admin then fail with
# "The uploaded file could not be moved to wp-content/uploads/...".
# RemoveHandler is the correct tool: it drops the mapping for this directory
# only.
<IfModule mod_mime.c>
    RemoveHandler .php .php3 .php4 .php5 .php7 .php8 .php8.0 .php8.1 .php8.2 .php8.3 .php8.4 .phtml .phps .phar .pht
</IfModule>

# Plus direct access protection. Script extensions only - images, PDFs and
# everything else stay reachable.
<FilesMatch "\.(php|php[0-9]|php[0-9]\.[0-9]|phtml|phps|pht|phar|shtml|cgi|pl|py|asp|aspx)$">
    Require all denied
</FilesMatch>

# Double extensions ("image.php.jpg") are still executed as PHP on some
# servers - so block any script extension ANYWHERE in the name.
<FilesMatch "\.(php|php[0-9]|phtml|phps|pht|phar|shtml|cgi|pl|py)\.">
    Require all denied
</FilesMatch>

# Only -Indexes: -ExecCGI and -Includes would require "AllowOverride Options",
# which many panels grant only through a whitelist - without it Apache returns
# 500 for the entire directory.
Options -Indexes
